const STR=['A','E','C','G'], FC=4, PM=[3,5,7,10,12,15], DM=12;

const SW=200,SH=110,ML=32,MR=10,MT=12,MB=28;

const BW=SW-ML-MR,BH=SH-MT-MB,CW=BW/FC,RH=BH/3;


/* ════════════════════════════════

   데이터

   chords: 팔레트에 있는 코드 목록

   rows:   행 배열 — 각 행은 measures 4개

   measure: { chordKey, lyric }

════════════════════════════════ */

let chords = [];   // {key, name, hasDiag, dots, stringStates, startFret}

let rows = [];     // [{measures:[{chordKey,lyric},...]}]


// localStorage에서 코드 빌더 데이터 불러오기 시도

function loadFromBuilder(){

  try{

    const raw = localStorage.getItem('uke_coll');

    if(raw){

      const coll = JSON.parse(raw);

      coll.forEach(e=>{

        chords.push({

          key: e.id,

          name: (e.root||'')+(e.qual||'')+(e.tensions&&e.tensions.length?' '+e.tensions.join(' '):''),

          hasDiag: true,

          dots: e.dots, stringStates: e.stringStates, startFret: e.startFret

        });

      });

    }

  }catch(e){}

}


/* ════════════════════════════════

   SVG 다이어그램 (인라인 소형)

════════════════════════════════ */

function mkSVG(dots, ss2, fr, w, h){

  const sx=w/SW,sy=h/SH,ml=ML*sx,mt=MT*sy;

  const bw=w-ml-MR*sx,bh=h-mt-MB*sy,cw=bw/FC,rh=bh/3;

  const dr=7*sx,nw=4.5*sx;

  const SY=s=>mt+s*rh, FCX=f=>ml+(f-.5)*cw, FLX=f=>ml+f*cw;

  let o=`<svg width="${w}" height="${h}" viewBox="0 0 ${w} ${h}" xmlns="http://www.w3.org/2000/svg">`;

  o+=`<rect width="${w}" height="${h}" fill="white"/>`;

  if(fr===1) o+=`<rect x="${ml-nw}" y="${mt}" width="${nw}" height="${bh}" rx="1" fill="#111"/>`;

  else o+=`<text x="${ml-2*sx}" y="${mt+bh*.5}" text-anchor="end" font-family="monospace" font-size="${7*sx}" fill="#888">${fr}fr</text>`;

  // 스트링

  for(let s=0;s<4;s++){const y=SY(s);o+=`<line x1="${ml}" y1="${y}" x2="${ml+bw}" y2="${y}" stroke="#222" stroke-width="${1.8*sx}"/>`;}

  // 프렛

  for(let f=1;f<=FC;f++) o+=`<line x1="${FLX(f)}" y1="${mt}" x2="${FLX(f)}" y2="${mt+bh}" stroke="#666" stroke-width="${1.5*sx}"/>`;

  o+=`<line x1="${ml}" y1="${mt}" x2="${ml}" y2="${mt+bh}" stroke="#666" stroke-width="${0.8*sx}"/>`;

  // open/mute

  for(let s=0;s<4;s++){

    const y=SY(s),st=ss2[s],sx2=ml-nw-4*sx;

    if(st==='open') o+=`<circle cx="${sx2}" cy="${y}" r="${dr*.44}" fill="none" stroke="#3a7a3a" stroke-width="1.2"/>`;

    else if(st==='mute'){const d=dr*.36;o+=`<line x1="${sx2-d}" y1="${y-d}" x2="${sx2+d}" y2="${y+d}" stroke="#cc3333" stroke-width="1.2"/><line x1="${sx2+d}" y1="${y-d}" x2="${sx2-d}" y2="${y+d}" stroke="#cc3333" stroke-width="1.2"/>`;}

  }

  // 포지션 마커

  for(let f=1;f<=FC;f++){

    const af=fr+f-1; if(!PM.includes(af)) continue;

    const x=FCX(f),y=mt+bh+18*sy,r=dr*.32;

    if(af===DM){const sp=dr*.5;o+=`<circle cx="${x-sp}" cy="${y}" r="${r}" fill="#b0a090"/><circle cx="${x+sp}" cy="${y}" r="${r}" fill="#b0a090"/>`;}

    else o+=`<circle cx="${x}" cy="${y}" r="${r}" fill="#b0a090"/>`;

  }

  // 바레

  dots.filter(d=>d.t==='barre').forEach(d=>{

    const x=FCX(d.f),y1=SY(d.s),y2=SY(d.e);

    o+=`<rect x="${x-dr}" y="${y1-dr}" width="${dr*2}" height="${y2-y1+dr*2}" rx="${dr}" fill="#111"/>`;

    o+=`<text x="${x}" y="${(y1+y2)/2+3*sy}" text-anchor="middle" font-family="monospace" font-size="${7*sx}" fill="white" font-weight="700">${fr+d.f-1}</text>`;

  });

  [{t:'dot',f:'#111'},{t:'tension',f:'#b8620a'}].forEach(({t,f})=>{

    dots.filter(d=>d.t===t).forEach(d=>{

      const x=FCX(d.f),y=SY(d.s);

      o+=`<circle cx="${x}" cy="${y}" r="${dr}" fill="${f}"/>`;

      o+=`<text x="${x}" y="${y+3*sy}" text-anchor="middle" font-family="monospace" font-size="${7*sx}" fill="white" font-weight="700">${fr+d.f-1}</text>`;

    });

  });

  return o+'</svg>';

}


/* ════════════════════════════════

   Canvas 다이어그램 (이미지 생성용)

════════════════════════════════ */

function drawDiagCanvas(chord, w, h){

  const cv=document.createElement('canvas'); cv.width=w; cv.height=h;

  const c=cv.getContext('2d');

  const sx=w/SW,sy=h/SH,ml=ML*sx,mt=MT*sy;

  const bw=w-ml-MR*sx,bh=h-mt-MB*sy,cw=bw/FC,rh=bh/3;

  const dr=7*sx,nw=4.5*sx;

  const fr=chord.startFret||1;

  const SY=s=>mt+s*rh, FCX=f=>ml+(f-.5)*cw, FLX=f=>ml+f*cw;

  c.fillStyle='#fff'; c.fillRect(0,0,w,h);

  if(fr===1){c.fillStyle='#111'; rrC(c,ml-nw,mt,nw,bh,1.5);c.fill();}

  else{c.fillStyle='#888';c.font=`${7*sx}px monospace`;c.textAlign='right';c.textBaseline='middle';c.fillText(fr+'fr',ml-2*sx,mt+bh*.5);}

  c.strokeStyle='#222';c.lineWidth=1.8*sx;

  for(let s=0;s<4;s++){const y=SY(s);c.beginPath();c.moveTo(ml,y);c.lineTo(ml+bw,y);c.stroke();}

  c.strokeStyle='#666';c.lineWidth=1.5*sx;

  for(let f=1;f<=FC;f++){const x=FLX(f);c.beginPath();c.moveTo(x,mt);c.lineTo(x,mt+bh);c.stroke();}

  c.lineWidth=0.8*sx;c.beginPath();c.moveTo(ml,mt);c.lineTo(ml,mt+bh);c.stroke();

  for(let s=0;s<4;s++){

    const y=SY(s),st=(chord.stringStates||[])[s]||'none',sx2=ml-nw-4*sx;

    if(st==='open'){c.strokeStyle='#3a7a3a';c.lineWidth=1.2;c.beginPath();c.arc(sx2,y,dr*.44,0,Math.PI*2);c.stroke();}

    else if(st==='mute'){const d=dr*.36;c.strokeStyle='#cc3333';c.lineWidth=1.2;c.beginPath();c.moveTo(sx2-d,y-d);c.lineTo(sx2+d,y+d);c.stroke();c.beginPath();c.moveTo(sx2+d,y-d);c.lineTo(sx2-d,y+d);c.stroke();}

  }

  for(let f=1;f<=FC;f++){

    const af=fr+f-1;if(!PM.includes(af))continue;

    const x=FCX(f),y=mt+bh+18*sy,r=dr*.32;c.fillStyle='#b0a090';

    if(af===DM){const sp=dr*.5;c.beginPath();c.arc(x-sp,y,r,0,Math.PI*2);c.fill();c.beginPath();c.arc(x+sp,y,r,0,Math.PI*2);c.fill();}

    else{c.beginPath();c.arc(x,y,r,0,Math.PI*2);c.fill();}

  }

  (chord.dots||[]).filter(d=>d.t==='barre').forEach(d=>{

    const x=FCX(d.f),y1=SY(d.s),y2=SY(d.e);

    c.fillStyle='#111';rrC(c,x-dr,y1-dr,dr*2,y2-y1+dr*2,dr);c.fill();

    c.fillStyle='#fff';c.font=`bold ${7*sx}px monospace`;c.textAlign='center';c.textBaseline='middle';c.fillText(fr+d.f-1,x,(y1+y2)/2);

  });

  [{t:'dot',f:'#111'},{t:'tension',f:'#b8620a'}].forEach(({t,f})=>{

    (chord.dots||[]).filter(d=>d.t===t).forEach(d=>{

      const x=FCX(d.f),y=SY(d.s);

      c.fillStyle=f;c.beginPath();c.arc(x,y,dr,0,Math.PI*2);c.fill();

      c.fillStyle='#fff';c.font=`bold ${7*sx}px monospace`;c.textAlign='center';c.textBaseline='middle';c.fillText(fr+d.f-1,x,y);

    });

  });

  return cv;

}

function rrC(ctx,x,y,w,h,r){

  ctx.beginPath();ctx.moveTo(x+r,y);ctx.lineTo(x+w-r,y);

  ctx.quadraticCurveTo(x+w,y,x+w,y+r);ctx.lineTo(x+w,y+h-r);

  ctx.quadraticCurveTo(x+w,y+h,x+w-r,y+h);ctx.lineTo(x+r,y+h);

  ctx.quadraticCurveTo(x,y+h,x,y+h-r);ctx.lineTo(x,y+r);

  ctx.quadraticCurveTo(x,y,x+r,y);ctx.closePath();

}

/* ════════════════════════════════

   행/마디 관리

   measure: { chordKeys:[], lyric:'' }

   한 마디에 코드 여러 개 가능

════════════════════════════════ */

function newMeasure(){ return {chordKeys:[''], lyric:''}; }


function addRow(){

  rows.push({measures:[newMeasure(),newMeasure(),newMeasure(),newMeasure()]});

  rendRows();

}

function delRow(ri){ rows.splice(ri,1); rendRows(); }

function addMeasureToRow(ri){ rows[ri].measures.push(newMeasure()); rendRows(); }

function delMeasure(ri,mi){

  rows[ri].measures.splice(mi,1);

  if(!rows[ri].measures.length) rows.splice(ri,1);

  rendRows();

}

function addChordToMeasure(ri,mi){ rows[ri].measures[mi].chordKeys.push(''); rendRows(); }

function removeChordFromMeasure(ri,mi,ci){

  rows[ri].measures[mi].chordKeys.splice(ci,1);

  if(!rows[ri].measures[mi].chordKeys.length) rows[ri].measures[mi].chordKeys=[''];

  rendRows();

}

async function genPreview(){

  if(!rows.length){alert('악보가 없습니다.');return;}

  const pov=document.getElementById('pov');

  const pmsg=document.getElementById('pmsg');

  pov.classList.add('show'); pmsg.textContent='악보 생성 중...';

  await tick();


  // A4 @ 150dpi: 1240 × 1754px

  const PW=1240, PH=1754, SCALE=1;

  const LM=52, RM=52;          // 좌우 마진

  const TM=80, BM=60;          // 상하 마진

  const TITLE_H=80;             // 제목 영역

  const INNER_W=PW-LM-RM;


  // 행 레이아웃

  const DIAG_W=56, DIAG_H=32;  // 다이어그램 크기

  const ABOVE=58;               // 기준선 위 (코드명+다이어그램)

  const BELOW=26;               // 기준선 아래 (가사)

  const ROW_H=ABOVE+BELOW+6;

  const ROW_GAP=16;


  const title=document.getElementById('scoreTitle').value||'우쿨렐레 악보';

  const artist=document.getElementById('scoreArtist').value||'';


  const cv=document.createElement('canvas');

  cv.width=PW; cv.height=PH;

  const c=cv.getContext('2d');


  // 배경

  c.fillStyle='#fff'; c.fillRect(0,0,PW,PH);


  // 제목 — 가운데

  c.fillStyle='#1a1410';

  c.font=`bold 32px Georgia,serif`;

  c.textAlign='center'; c.textBaseline='middle';

  c.fillText(title, PW/2, TM+24);


  // 아티스트 — 오른쪽, 약간 여백(30px)

  if(artist){

    c.font=`17px -apple-system,Helvetica Neue,Arial,sans-serif`;

    c.fillStyle='#5a4a3a';

    c.textAlign='right';

    c.textBaseline='middle';

    c.fillText(artist, PW-120, TM+56);

  }

  // 제목 아래 실선 없음


  const startY=TM+TITLE_H;


  for(let ri=0;ri<rows.length;ri++){

    pmsg.textContent=`행 ${ri+1}/${rows.length} 렌더링...`;

    await tick();


    const rowY=startY+ri*(ROW_H+ROW_GAP);

    if(rowY+ROW_H > PH-BM) break; // 페이지 넘침 방지


    const lineY=rowY+ABOVE;

    const measures=rows[ri].measures;

    const N=measures.length||1;


    // ── 마디 너비: INNER_W를 N등분 (완전 균등) ──

    const mW=INNER_W/N;


    // 기준선

    c.strokeStyle='#1a1410'; c.lineWidth=1.8;

    c.beginPath();c.moveTo(LM,lineY);c.lineTo(LM+INNER_W,lineY);c.stroke();


    // 왼쪽 세로선

    c.strokeStyle='#1a1410'; c.lineWidth=2.5;

    c.beginPath();c.moveTo(LM,lineY-ABOVE+4);c.lineTo(LM,lineY+14);c.stroke();


    for(let mi=0;mi<N;mi++){

      const m=measures[mi];

      if(!m.chordKeys) m.chordKeys=[m.chordKey||''];

      const mx=LM+mi*mW;


      // 마디 내 코드들

      const validKeys=m.chordKeys.filter(k=>k);

      const nCodes=validKeys.length||1;

      const codeSlotW=mW/nCodes;


      validKeys.forEach((ck,ci)=>{

        const ch=chords.find(c2=>c2.key===ck);

        if(!ch) return;

        const cx=mx+ci*codeSlotW;

        const cname=ch.name.split(' ')[0];

        const tens=ch.name.split(' ').slice(1).join(' ');


        // 코드명

        c.fillStyle='#1a1410';

        c.font=`bold 15px Georgia,serif`;

        c.textAlign='left'; c.textBaseline='bottom';

        c.fillText(cname, cx+4, lineY-2);


        // 텐션

        if(tens){

          const cnW=c.measureText(cname).width;

          c.fillStyle='#b8620a';

          c.font=`bold 8px monospace`;

          c.fillText(tens, cx+4+cnW+2, lineY-10);

        }


        // 다이어그램 — 코드명 위, 기준선 위에만

        if(ch.hasDiag){

          c.font=`bold 15px Georgia,serif`;

          const cnW=c.measureText(cname).width;

          const diagX=cx+4+cnW+6;

          const diagY=lineY-DIAG_H-2;

          // 다이어그램이 다음 코드 슬롯을 넘지 않도록

          const maxDW=Math.min(DIAG_W, codeSlotW-cnW-20);

          if(maxDW>20){

            const dRatio=DIAG_H/DIAG_W;

            const dcv=drawDiagCanvas(ch, maxDW*2, maxDW*2*dRatio);

            c.drawImage(dcv, diagX, diagY, maxDW, maxDW*dRatio);

          }

        }

      });


      // 코드 없는 마디도 가사는 표시

      if(m.lyric){

        c.fillStyle='#1a1410';

        c.font=`14px -apple-system,Helvetica Neue,Arial,sans-serif`;

        c.textAlign='left'; c.textBaseline='top';

        c.fillText(m.lyric, mx+4, lineY+8);

      }


      // 마디 세로선

      c.strokeStyle='#1a1410';

      c.lineWidth=mi===N-1 ? 2.5 : 1.2;

      c.beginPath();

      c.moveTo(mx+mW, lineY-ABOVE+4);

      c.lineTo(mx+mW, lineY+14);

      c.stroke();

    }

  }


  pov.classList.remove('show');


  const dataUrl=cv.toDataURL('image/jpeg',0.95);

  const panel=document.getElementById('previewPanel');

  panel.classList.add('show');

  const old=panel.querySelector('img');if(old)old.remove();

  const img=new Image();img.src=dataUrl;

  img.style.cssText='max-width:100%;border-radius:6px;border:1px solid #d8cfc0;display:block;margin-top:4px';

  panel.appendChild(img);

  img.scrollIntoView({behavior:'smooth'});

}

function rendMiniSvg(){

  const fr=parseInt(document.getElementById('miniStartFret').value)||1;

  const wrap=document.getElementById('miniSvgWrap');

  wrap.innerHTML=mkSVG(miniDots,miniSS,fr,160,90);

  const svg=wrap.querySelector('svg');

  svg.style.cursor='crosshair';

  svg.addEventListener('click',onMiniClick);

}


function onMiniClick(ev){

  const svg=ev.currentTarget;

  const r=svg.getBoundingClientRect();

  const px=(ev.clientX-r.left)*(SW/r.width);

  const py=(ev.clientY-r.top)*(SH/r.height);

  // hit test

  if(px<ML||px>ML+BW) return;

  const f=Math.floor((px-ML)/CW)+1;

  if(f<1||f>FC) return;

  let bestS=-1,md=Infinity;

  for(let s=0;s<4;s++){const d=Math.abs(py-(MT+s*RH));if(d<md){md=d;bestS=s;}}

  if(md>RH*.7) return;

  const cell={s:bestS,f};

  if(miniMode==='erase'){

    miniDots=miniDots.filter(d=>!(d.s===cell.s&&d.f===cell.f));

  } else {

    const idx=miniDots.findIndex(d=>d.s===cell.s&&d.f===cell.f);

    if(idx>=0) miniDots.splice(idx,1);

    else miniDots.push({s:cell.s,f:cell.f,t:miniMode==='tension'?'tension':'dot'});

  }

  rendMiniSvg();

}


function clearMiniDots(){

  miniDots=[];miniSS=['none','none','none','none'];

  rendMiniSS();rendMiniSvg();

}

}
